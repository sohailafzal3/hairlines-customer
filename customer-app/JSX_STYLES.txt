  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="dark-content" backgroundColor="#F8FAFC" />
      <LoadingOverlay visible={loading} />

      <KeyboardAvoidingView
        behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
        style={styles.keyboardView}
      >
        <ScrollView
          contentContainerStyle={styles.scrollContent}
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}
        >
          {/* Top Bar - Back Button */}
          <View style={styles.topBar}>
            <TouchableOpacity
              onPress={handleBack}
              style={styles.backButton}
              activeOpacity={0.8}
            >
              <Ionicons name="arrow-back" size={20} color="#1E293B" />
            </TouchableOpacity>
          </View>

          {/* Header Title */}
          <View style={styles.header}>
            <Text style={styles.title}>Complete Profile</Text>
            <Text style={styles.subtitle}>
              Help us personalize your customer profile & services
            </Text>
          </View>

          {/* Avatar Upload Container */}
          <View style={styles.avatarSection}>
            <TouchableOpacity
              onPress={pickImage}
              activeOpacity={0.85}
              style={styles.avatarWrapper}
            >
              {profileImage ? (
                <Image source={{ uri: profileImage }} style={styles.profileImage} />
              ) : (
                <View style={styles.avatarPlaceholder}>
                  <Ionicons name="person" size={40} color="#94A3B8" />
                </View>
              )}
              {uploadingImage ? (
                <View style={styles.uploadingOverlay}>
                  <ActivityIndicator size="small" color="#FFFFFF" />
                </View>
              ) : (
                <View style={styles.cameraBadge}>
                  <Ionicons name="camera" size={14} color="#FFFFFF" />
                </View>
              )}
            </TouchableOpacity>
            <Text style={styles.avatarLabel}>
              {uploadingImage ? 'Uploading photo...' : profileImage ? 'Change Photo' : 'Upload Photo'}
            </Text>
          </View>

          <View style={styles.formContainer}>
            
            {/* PERSONAL INFO CARD */}
            <Text style={styles.sectionHeader}>Personal Info</Text>
            <View style={styles.card}>
              <View style={styles.rowFields}>
                <Input
                  label="First Name"
                  placeholder="First Name"
                  value={firstName}
                  onChangeText={setFirstName}
                  autoCapitalize="words"
                  containerStyle={styles.flexHalf}
                />
                <Input
                  label="Last Name"
                  placeholder="Last Name"
                  value={lastName}
                  onChangeText={setLastName}
                  autoCapitalize="words"
                  containerStyle={styles.flexHalf}
                />
              </View>

              <Input
                label="Email Address"
                placeholder="customer@example.com"
                value={email}
                onChangeText={setEmail}
                keyboardType="email-address"
                autoCapitalize="none"
              />

              <Text style={styles.inputLabel}>Gender</Text>
              <View style={styles.genderSegmentRow}>
                {[
                  { key: 'Male', label: 'Male', icon: 'male' },
                  { key: 'Female', label: 'Female', icon: 'female' },
                ].map((item) => {
                  const isActive = gender === item.key;
                  return (
                    <TouchableOpacity
                      key={item.key}
                      style={[styles.genderCard, isActive && styles.genderCardActive]}
                      onPress={() => setGender(item.key)}
                      activeOpacity={0.8}
                    >
                      <Ionicons
                        name={item.icon as any}
                        size={16}
                        color={isActive ? '#222D63' : '#64748B'}
                        style={{ marginRight: 6 }}
                      />
                      <Text
                        style={[
                          styles.genderCardText,
                          isActive && styles.genderCardTextActive,
                        ]}
                      >
                        {item.label}
                      </Text>
                    </TouchableOpacity>
                  );
                })}
              </View>

              <Text style={styles.inputLabel}>Date of Birth</Text>
              <TouchableOpacity
                style={styles.datePickerTrigger}
                onPress={openDobModal}
                activeOpacity={0.8}
              >
                <Text
                  style={[
                    styles.datePickerText,
                    !dob && { color: '#94A3B8' },
                  ]}
                >
                  {dob ? `${dob.slice(4, 6)}/${dob.slice(6, 8)}/${dob.slice(0, 4)}` : 'Select Date of Birth'}
                </Text>
                <Ionicons
                  name="calendar-outline"
                  size={18}
                  color="#222D63"
                />
              </TouchableOpacity>
            </View>

            {/* SECURITY CARD */}
            <Text style={styles.sectionHeader}>Security</Text>
            <View style={styles.card}>
              <Input
                label="Password"
                placeholder="Create password"
                value={password}
                onChangeText={setPassword}
                leftIcon="lock-closed-outline"
                secureTextEntry
              />

              <Input
                label="Confirm Password"
                placeholder="Re-enter password"
                value={confirmPassword}
                onChangeText={setConfirmPassword}
                leftIcon="lock-closed-outline"
                secureTextEntry
              />
              {confirmPassword.length > 0 && (
                <View style={styles.passwordMatchNotice}>
                  <Ionicons
                    name={password === confirmPassword ? 'checkmark-circle' : 'alert-circle'}
                    size={14}
                    color={password === confirmPassword ? '#10B981' : '#EF4444'}
                    style={{ marginRight: 5 }}
                  />
                  <Text
                    style={[
                      styles.passwordMatchText,
                      { color: password === confirmPassword ? '#059669' : '#DC2626' },
                    ]}
                  >
                    {password === confirmPassword ? 'Passwords match' : 'Passwords do not match'}
                  </Text>
                </View>
              )}
            </View>

            {/* ADDRESS CARD */}
            <Text style={styles.sectionHeader}>Address</Text>
            <View style={styles.card}>
              <Text style={styles.inputLabel}>Service / Residence Address</Text>
              <TouchableOpacity
                onPress={() => {
                  setAddressSearchQuery(address);
                  setAddressModalVisible(true);
                }}
                activeOpacity={0.8}
                style={styles.addressTrigger}
              >
                <Ionicons
                  name="location-outline"
                  size={18}
                  color="#222D63"
                  style={{ marginRight: 10 }}
                />
                <Text
                  style={[
                    styles.addressTriggerText,
                    !address && { color: '#94A3B8' },
                  ]}
                  numberOfLines={2}
                >
                  {address || 'Tap to search or set location'}
                </Text>
                <Ionicons name="chevron-forward" size={16} color="#94A3B8" />

              <TouchableOpacity
                style={styles.gpsQuickDetectBtn}
                onPress={handleUseCurrentLocation}
                disabled={detectingGps}
                activeOpacity={0.7}
              >
                {detectingGps ? (
                  <ActivityIndicator size="small" color="#222D63" style={{ marginRight: 8 }} />
                ) : (
                  <Ionicons name="navigate-circle" size={18} color="#222D63" style={{ marginRight: 8 }} />
                )}
                <Text style={styles.gpsQuickDetectText}>
                  {detectingGps ? 'Detecting current GPS location...' : 'Use My Current GPS Location'}
                </Text>
              </TouchableOpacity>

              <View style={styles.rowFields}>
                <Input
                  label="Apt / Suite / Unit #"
                  placeholder="e.g. Apt 4B"
                  value={streetAddressLine1}
                  onChangeText={setStreetAddressLine1}
                  containerStyle={styles.flexHalf}
                />
                <Input
                  label="Referral Code"
                  placeholder="e.g. FRIEND20"
                  value={referralCode}
                  onChangeText={setReferralCode}
                  autoCapitalize="characters"
                  containerStyle={styles.flexHalf}
                />
              </View>

              <View style={styles.rowFields}>
                <Input
                  label="City"
                  placeholder="City"
                  value={city}
                  onChangeText={setCity}
                  containerStyle={styles.flexHalf}
                />
                <Input
                  label="State"
                  placeholder="State"
                  value={state}
                  onChangeText={setState}
                  containerStyle={styles.flexHalf}
                />
              </View>

              <View style={styles.rowFields}>
                <Input
                  label="Postal Code"
                  placeholder="e.g. 33825"
                  value={postalCode}
                  onChangeText={setPostalCode}
                  keyboardType="numeric"
                  maxLength={10}
                  containerStyle={styles.flexHalf}
                />
                <Input
                  label="Country"
                  placeholder="USA"
                  value={country}
                  onChangeText={setCountry}
                  containerStyle={styles.flexHalf}
                />
              </View>

              <Text style={styles.inputLabel}>Accessibility Category</Text>
              <View style={styles.accessibilityRow}>
                {[
                  { id: 0, label: 'Standard' },
                  { id: 1, label: 'Mental' },
                  { id: 2, label: 'Physical' },
                ].map((item) => (
                  <TouchableOpacity
                    key={item.id}
                    style={[
                      styles.accessibilityCard,
                      isPhysicallyDisabled === item.id && styles.accessibilityCardActive,

                    <View
                      style={[
                        styles.radioCircle,
                        isPhysicallyDisabled === item.id && styles.radioCircleActive,

                    >
                      {isPhysicallyDisabled === item.id && <View style={styles.radioInnerCircle} />}
                    </View>
                    <Text
                      style={[
                        styles.accessibilityCardText,
                        isPhysicallyDisabled === item.id && styles.accessibilityCardTextActive,
                      ]}
                    >
                      {item.label}
                    </Text>
                  </TouchableOpacity>

              </View>
            </View>

            {/* Agreement Checkbox & Text */}
            <View style={styles.agreementRow}>
              <TouchableOpacity
                onPress={() => setIsTermsAccepted(!isTermsAccepted)}

                    onPress={() => setTempMonth((m) => (m < 12 ? m + 1 : 1))}

                    <Ionicons name="chevron-down" size={20} color="#1E293B" />
                  </TouchableOpacity>
                </View>

                <View style={styles.selectorBox}>
                  <TouchableOpacity
                    onPress={() => setTempDay((d) => (d > 1 ? d - 1 : 31))}

                    <Ionicons name="chevron-up" size={20} color="#1E293B" />

                  <Text style={styles.selectorVal}>{tempDay}</Text>

                  <TouchableOpacity

                      setTempYear((y) =>
                        y < new Date().getFullYear() - 13 ? y + 1 : y

                    onPress={() =>
                      setTempYear((y) => (y > 1930 ? y - 1 : y))

                    style={styles.arrowBtn}
                  >
                    <Ionicons name="chevron-down" size={20} color="#1E293B" />
                  </TouchableOpacity>
                </View>
              </View>
            </View>

            <View style={styles.modalFooter}>
              <Button title="Apply Date" onPress={handleApplyDOB} />
            </View>
          </View>
        </View>
      </Modal>

      {/* Address Selection Modal */}
      <Modal
        visible={addressModalVisible}
        animationType="slide"
        transparent={false}
        onRequestClose={() => setAddressModalVisible(false)}
      >
        <SafeAreaView style={styles.addressModalContainer}>
          <View style={styles.addressModalHeader}>
            <TouchableOpacity
              onPress={() => setAddressModalVisible(false)}
              style={styles.modalCloseBtn}
            >
              <Ionicons name="arrow-back" size={22} color="#1E293B" />
            </TouchableOpacity>
            <Text style={styles.addressModalTitle}>Set Address</Text>
            <View style={{ width: 36 }} />

          <View style={styles.searchWrapper}>
            <View style={styles.searchBarWrap}>
              <Ionicons
                name="search"
                size={18}
                color="#94A3B8"
                style={{ marginRight: 8 }}

              <TextInput
                placeholder="Search street, city, zip code..."
                placeholderTextColor="#94A3B8"

                autoFocus={true}

              />
              {searchingAddress && <ActivityIndicator size="small" color="#222D63" />}

          {/* Quick Preset Options */}

            <Text style={styles.quickOptionsTitle}>Quick Suggestions</Text>
            <TouchableOpacity
              style={styles.quickOptionRow}
              onPress={handleUseCurrentLocation}
              activeOpacity={0.8}
            >
              <View style={[styles.quickOptionIconWrap, { backgroundColor: '#F8FAFC' }]}>
                <Ionicons name="locate" size={18} color="#222D63" />
              </View>
              <View style={{ flex: 1 }}>
                <Text style={styles.quickOptionLabel}>Use Current Location</Text>
                <Text style={styles.quickOptionSub}>Auto-detect via GPS</Text>
              </View>
              <Ionicons name="chevron-forward" size={16} color="#94A3B8" />
            </TouchableOpacity>
          </View>

          {/* Prediction Results */}
          <ScrollView
            style={styles.predictionsScroll}
            contentContainerStyle={styles.predictionsScrollContent}
            keyboardShouldPersistTaps="handled"
          >
            {addressPredictions.map((pred, index) => (
              <TouchableOpacity
                key={pred.place_id || index.toString()}
                style={styles.predictionItem}
                onPress={() => selectPlacePrediction(pred)}
                activeOpacity={0.7}
              >
                <Ionicons
                  name="location-outline"
                  size={18}
                  color="#222D63"
                  style={{ marginRight: 12, marginTop: 2 }}
                />
                <View style={{ flex: 1 }}>
                  <Text style={styles.predictionText}>{pred.description}</Text>
                  {(pred.city || pred.state) && (
                    <Text style={styles.predictionSubText}>
                      {[pred.city, pred.state, pred.country].filter(Boolean).join(', ')}
                    </Text>
                  )}
                </View>
                <Ionicons name="chevron-forward" size={14} color="#94A3B8" />
              </TouchableOpacity>
            ))}

            {!searchingAddress && addressSearchQuery.length >= 2 && addressPredictions.length === 0 && (
              <View style={styles.noResultsBox}>
                <Ionicons name="map-outline" size={36} color="#94A3B8" />
                <Text style={styles.noResultsTitle}>No locations found</Text>
                <Text style={styles.noResultsSub}>Try entering a street name, city, or postal code.</Text>
                <TouchableOpacity
                  style={styles.useEnteredTextBtn}
                  onPress={() => {
                    setAddress(addressSearchQuery);
                    setAddressModalVisible(false);
                  }}
                >
                  <Text style={styles.useEnteredTextBtnLabel}>Use "{addressSearchQuery}" anyway</Text>
                </TouchableOpacity>
              </View>
            )}
          </ScrollView>
        </SafeAreaView>
      </Modal>

      {/* Terms & Conditions Modal */}
      <Modal
        visible={termsModalVisible}
        animationType="slide"
        transparent={false}
        onRequestClose={() => setTermsModalVisible(false)}
      >
        <SafeAreaView style={styles.modalContainer}>
          <View style={styles.modalHeader}>
            <Text style={styles.modalTitle}>Terms & Conditions</Text>
            <TouchableOpacity
              onPress={() => setTermsModalVisible(false)}
              style={styles.modalCloseBtn}
            >
              <Ionicons name="close" size={24} color="#1E293B" />
            </TouchableOpacity>
          </View>
          <ScrollView
            style={styles.modalScroll}
            contentContainerStyle={styles.modalScrollContent}
            showsVerticalScrollIndicator={true}
          >
            {termsLoading ? (
              <View style={styles.termsLoadingContainer}>
                <ActivityIndicator size="large" color="#222D63" />
                <Text style={styles.termsLoadingText}>Loading latest terms...</Text>
              </View>
            ) : (
              <Text style={styles.modalBodyText}>
                {termsDescription
                  ? termsDescription.replace(/<[^>]+>/g, '').trim()
                  : 'Welcome to Hairlines. By registering and booking services through our platform, you agree to treat our verified service professionals with respect, adhere to safety and hygiene protocols during appointments, maintain accurate booking and contact information, and comply with our transparent cancellation and refund policy.'}
              </Text>
            )}
          </ScrollView>
          <View style={styles.modalFooter}>
            <Button
              title="I Understand & Accept"
              onPress={() => {
                setIsTermsAccepted(true);
                setTermsModalVisible(false);
              }}
            />
          </View>
        </SafeAreaView>
      </Modal>
    </SafeAreaView>
  );
}
